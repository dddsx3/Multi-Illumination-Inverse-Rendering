#### §3.1 thermal_resume_guard.py 伪代码

```python
# tests_audit/thermal_resume_guard.py（执行者落地到 repo/）
import subprocess, time, sys

def read_gpu_max_temp():
    out = subprocess.check_output(
        ['nvidia-smi', '--query-gpu=temperature.gpu',
         '--format=csv,noheader,nounits'],
        text=True)
    return max(int(x.strip()) for x in out.strip().splitlines())

def wait_until_cool(resume_c=75, n_samples=5, interval_s=60, log_path=None):
    """5/5 次读数 ≤ resume_c 才返回 True；任一 > resume_c 立即停。"""
    passed = 0
    while passed < n_samples:
        t = read_gpu_max_temp()
        line = f"{time.time():.0f} {t}"
        if log_path:
            with open(log_path, 'a') as f:
                f.write(line + '\n')
        if t > resume_c:
            print(f"[thermal] 读数 {t}°C > {resume_c}°C，冷却 10 分钟...")
            time.sleep(600)
            passed = 0
            continue
        passed += 1
        if passed < n_samples:
            time.sleep(interval_s)
    return True

if __name__ == "__main__":
    if wait_until_cool():
        print("[thermal] 5/5 通过，可以续跑")
        sys.exit(0)
    else:
        print("[thermal] 5 次内有 >75°C，抛 ThermalNotReady")
        sys.exit(42)  # 与 thermal_guard 退出码一致
```
